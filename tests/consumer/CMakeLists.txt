cmake_minimum_required(VERSION 3.21)
project(SpatialWebBrowserInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(SpatialWebBrowser 0.1.0 EXACT REQUIRED CONFIG)
add_executable(spatial-web-consumer main.cpp)
target_link_libraries(spatial-web-consumer PRIVATE SpatialWeb::Browser)

option(SPATIAL_WEB_TEST_COMPOSITION "Exercise the installed composition SDK" ON)
if(SPATIAL_WEB_TEST_COMPOSITION)
    find_package(SpatialWebComposition 0.1.0 EXACT REQUIRED CONFIG)
    target_link_libraries(spatial-web-consumer PRIVATE SpatialWeb::Composition)
    target_compile_definitions(spatial-web-consumer PRIVATE SPATIAL_WEB_TEST_COMPOSITION)
endif()

option(SPATIAL_WEB_TEST_PROJECTION "Exercise the installed spatial browser projection SDK" ${SPATIAL_WEB_TEST_COMPOSITION})
if(SPATIAL_WEB_TEST_PROJECTION)
    find_package(SpatialWebProjection 0.1.0 EXACT REQUIRED CONFIG)
    target_link_libraries(spatial-web-consumer PRIVATE SpatialWeb::Projection)
    target_compile_definitions(spatial-web-consumer PRIVATE SPATIAL_WEB_TEST_PROJECTION)
endif()
